cmake_minimum_required(VERSION 3.22)
project(j2p_windows_native_host LANGUAGES C)
include(CTest)
get_filename_component(J2P_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(SESSION_DIR "${J2P_ROOT}/ports/native-host/common")
add_subdirectory("${J2P_ROOT}/ports/native-scene-matrices" "${CMAKE_BINARY_DIR}/scene")
set_target_properties(j2p_native_scene_matrices PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set_target_properties(j2p_scene_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_executable(j2p_native_host host.c "${SESSION_DIR}/host_session.c")
target_include_directories(j2p_native_host PRIVATE "${SESSION_DIR}" "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(j2p_native_host PRIVATE j2p_native_scene_matrices user32 gdi32)
set_target_properties(j2p_native_host PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_compile_options(j2p_native_host PRIVATE -Wall -Wextra -Werror -Wpedantic)
add_test(NAME windows_native_two_windows COMMAND j2p_native_host --smoke)
